🧠 Module 3 Summary

  • Operators: assignment, arithmetic, logical, comparison, string
  • Precedence: controls evaluation order
  • User Interaction: alert, confirm, prompt

💻 Sample Code


// Arithmetic and assignment
let x = 10;
x += 5; // x = 15
console.log(x);

// Comparison
console.log(5 == "5");   // true
console.log(5 === "5");  // false

// Logical
let isSunny = true;
let isWeekend = false;
console.log(isSunny && isWeekend); // false

// User interaction
alert("Welcome to Module 3!");
let name = prompt("What's your name?");
let confirmEntry = confirm("Do you want to continue?");
console.log("User:", name, "Confirmed:", confirmEntry);
