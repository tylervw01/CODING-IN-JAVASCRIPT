💻 Your First Code


// Output a message
console.log("Hello, World!");

// Declare a variable
let message = "JavaScript is awesome!";
console.log(message);

// Simple math
let a = 5;
let b = 3;
let sum = a + b;
console.log("Sum:", sum);

// Conditional logic
if (sum > 5) {
  console.log("Sum is greater than 5");
} else {
  console.log("Sum is 5 or less");
}

// Loop example
for (let i = 1; i <= 3; i++) {
  console.log("Loop iteration:", i);
}

Open your browser console to see the output.
